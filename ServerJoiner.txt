-- SMOOTH OPEN & CLOSE ANIMATION LOGIC
local uiVisible = true
local openTweenInfo = TweenInfo.new(0.3, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
local closeTweenInfo = TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
local isAnimating = false

local function ToggleUI()
if isAnimating then return end
isAnimating = true
uiVisible = not uiVisible

if uiVisible then
MainFrame.Visible = true
TweenService:Create(MainFrame, openTweenInfo, {
BackgroundTransparency = 0,
Size = DefaultSize
}):Play()
TweenService:Create(MainUIStroke, openTweenInfo, { Transparency = 0 }):Play()
TweenService:Create(InputBox, openTweenInfo, { BackgroundTransparency = 0, TextTransparency = 0 }):Play()
TweenService:Create(InputStroke, openTweenInfo, { Transparency = 0 }):Play()
TweenService:Create(SoloButton, openTweenInfo, { BackgroundTransparency = 0, TextTransparency = 0 }):Play()

task.wait(0.3)
isAnimating = false
else
TweenService:Create(MainFrame, closeTweenInfo, {
BackgroundTransparency = 1,
Size = HiddenSize
}):Play()
TweenService:Create(MainUIStroke, closeTweenInfo, { Transparency = 1 }):Play()
TweenService:Create(InputBox, closeTweenInfo, { BackgroundTransparency = 1, TextTransparency = 1 }):Play()
TweenService:Create(InputStroke, closeTweenInfo, { Transparency = 1 }):Play()
TweenService:Create(SoloButton, closeTweenInfo, { BackgroundTransparency = 1, TextTransparency = 1 }):Play()

task.wait(0.2)
MainFrame.Visible = false
isAnimating = false
end
end

ToggleIconButton.MouseButton1Click:Connect(ToggleUI)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
if gameProcessed then return end
if input.KeyCode == Enum.KeyCode.RightControl or input.KeyCode == Enum.KeyCode.Insert then
ToggleUI()
end
end)

-- SERVER SEARCH & JOIN LOGIC
local isSearching = false

local function FindAndJoinServer()
if isSearching then return end
isSearching = true
SoloButton.Text = "Searching..."

local requestedNum = tonumber(InputBox.Text)
local targetPlayers = nil

if requestedNum then
if requestedNum > 6 then
targetPlayers = nil
else
targetPlayers = math.clamp(math.floor(requestedNum), 0, 6)
end
end

local targetServerId = nil
local closestPlayerCount = math.huge
local cursor = ""

for _ = 1, 4 do
local requestUrl = "https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/0?sortOrder=Asc&limit=100"
if cursor ~= "" then
requestUrl = requestUrl .. "&cursor=" .. cursor
end

local success, response = pcall(function()
return game:HttpGet(requestUrl)
end)

if success and response then
local data = HttpService:JSONDecode(response)
if data and data.data then
for _, server in ipairs(data.data) do
if server.playing and server.playing < server.maxPlayers and server.id ~= game.JobId then
if targetPlayers then
if server.playing == targetPlayers then
targetServerId = server.id
break
elseif server.playing < closestPlayerCount then
closestPlayerCount = server.playing
targetServerId = server.id
end
else
if server.playing < closestPlayerCount then
closestPlayerCount = server.playing
targetServerId = server.id
end
end
end
end
end

if targetServerId and targetPlayers and closestPlayerCount == targetPlayers then
break
end

if data and data.nextPageCursor then
cursor = data.nextPageCursor
else
break
end
else
break
end
end

if targetServerId then
SoloButton.Text = "Teleporting..."
TeleportService:TeleportToPlaceInstance(PlaceId, targetServerId, LocalPlayer)
else
SoloButton.Text = "No Server Found"
task.wait(2)
SoloButton.Text = "Join Server"
isSearching = false
end
end

SoloButton.MouseButton1Click:Connect(FindAndJoinServer)
